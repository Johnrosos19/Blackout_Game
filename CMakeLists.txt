cmake_minimum_required(VERSION 3.16)
project(BLACKOUT CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
add_executable(BLACKOUT main.cpp Shader.cpp Mesh.cpp Textures.cpp Scene.cpp LitTest.cpp Game.cpp Hud.cpp)
target_compile_definitions(BLACKOUT PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(BLACKOUT PRIVATE OpenGL::GL GLEW::GLEW glfw glm::glm)
set_target_properties(BLACKOUT PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
